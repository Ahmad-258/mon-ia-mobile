import threading
import json
import requests
from kivy.app import App
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.textinput import TextInput
from kivy.uix.button import Button
from kivy.uix.scrollview import ScrollView
from kivy.uix.label import Label
from kivy.uix.image import AsyncImage

INSTRUCTIONS_SYSTEME = (
    "Tu es l'IA personnelle d'Ahmad. Tu es un expert en programmation surpuissant. "
    "Tu codes sans faire d'erreurs, tu expliques clairement et tu l'aides sur ses grands projets. "
    "Si Ahmad te demande une image ou une photo (ex: 'dessine...', 'fais une photo de...'), "
    "commence TOUJOURS ta réponse exactement par le mot exact: suivi de la description en anglais de ce qu'il veut."
)

class ChatApp(App):
    def build(self):
        self.title = "L'IA d'Ahmad v2"
        self.layout = BoxLayout(orientation='vertical', padding=10, spacing=10)
        
        self.scroll_view = ScrollView(size_hint=(1, 0.85))
        self.chat_history = BoxLayout(orientation='vertical', size_hint_y=None, spacing=10)
        self.chat_history.bind(minimum_height=self.chat_history.setter('height'))
        self.scroll_view.add_widget(self.chat_history)
        self.layout.add_widget(self.scroll_view)
        
        self.input_layout = BoxLayout(orientation='horizontal', size_hint=(1, 0.15), spacing=10)
        self.text_input = TextInput(hint_text="Demande un code ou une image...", multiline=False, size_hint=(0.75, 1))
        self.text_input.bind(on_text_validate=self.envoyer_message)
        
        self.send_button = Button(text="Envoyer", size_hint=(0.25, 1), background_color=(0.2, 0.6, 1, 1))
        self.send_button.bind(on_press=self.envoyer_message)
        
        self.input_layout.add_widget(self.text_input)
        self.input_layout.add_widget(self.send_button)
        self.layout.add_widget(self.input_layout)
        return self.layout

    def ajouter_message(self, auteur, texte):
        lbl = Label(text=f"[b]{auteur}:[/b] {texte}", markup=True, size_hint_y=None, color=(1, 1, 1, 1))
        lbl.bind(texture_size=lambda instance, value: setattr(instance, 'height', value))
        lbl.text_size = (self.scroll_view.width - 20, None)
        self.chat_history.add_widget(lbl)
        self.scroll_view.scroll_y = 0

    def ajouter_image(self, url):
        img = AsyncImage(source=url, size_hint_y=None, height=300)
        self.chat_history.add_widget(img)
        self.scroll_view.scroll_y = 0

    def envoyer_message(self, instance):
        prompt = self.text_input.text.strip()
        if not prompt:
            return
        self.ajouter_message("Ahmad", prompt)
        self.text_input.text = ""
        threading.Thread(target=calculer_reponse, args=(self, prompt), daemon=True).start()

def calculer_reponse(app_instance, prompt_utilisateur):
    try:
        # URL officielle et compatible sans clés pour les requetes de chat complétion
        url = "https://huggingface.co"
        headers = {"Content-Type": "application/json"}
        payload = {
            "model": "Qwen/Qwen2.5-Coder-32B-Instruct",
            "messages": [
                {"role": "system", "content": INSTRUCTIONS_SYSTEME},
                {"role": "user", "content": prompt_utilisateur}
            ],
            "max_tokens": 1000
        }
        
        response = requests.post(url, headers=headers, json=payload, timeout=20)
        res_json = response.json()
        reponse = res_json["choices"][0]["message"]["content"]
        
        if reponse.strip().startswith(""):
            description_image = reponse.replace("", "").strip()
            app_instance.ajouter_message("Mon IA", "Je génère ta photo, un instant...")
            url_flux = f"https://pollinations.ai{description_image.replace(' ', '%20')}?width=512&height=512&seed=42"
            app_instance.ajouter_image(url_flux)
        else:
            app_instance.ajouter_message("Mon IA", reponse)
            
    except Exception as e:
        app_instance.ajouter_message("Système", "Erreur réseau. Vérifie ta connexion Internet.")

if __name__ == '__main__':
    ChatApp().run()
